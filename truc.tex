    fonction puissance(x : TYPE, n : entier long) TYPE
    debut
        si estNul(n) alors 
            retourner elementNeutre()
        sinon si estpair(n) alors
            retourner carre(puissance(x,n>>1))
        sinon
            retourner produit(x , carre(puissance(x,n>>1)))
    fin
