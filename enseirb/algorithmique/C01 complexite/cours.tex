\begin{definition}{}{domination asymptotique}
    Soit $(u_n)_{n \in \N}$ et $(v_n)_{n \in \N}$ deux suites réelles positives. On dit que \notion{$(u_n)_{n \in \N}$ est dominée par $v_n$} si~:
    $$\exists C > 0,\, \forall n \in \N,\, u_n \leq C v_n$$
    On note alors $\displaystyle u_n \symbolelim{n \to + \infty}{}{=} \mc{O}(v_n)$.
\end{definition}

\begin{definition}{}{comparabilité asymptotique}
    Soit $(u_n)_{n \in \N}$ et $(v_n)_{n \in \N}$ deux suites réelles positives. On dit que \notion{$(u_n)_{n \in \N}$ est comparable à $v_n$} si l'une peut toujours borner l'autre~:
    $$\exists C_1 > 0,\, \exists C_2 > 0,\, \forall n \in \N,\, C_1 v_n \leq u_n \leq C_2 v_n$$
    On note alors $\displaystyle u_n \symbolelim{n \to + \infty}{}{=} \Theta(v_n)$.
\end{definition}

\begin{remarque}{}{}
    Malgré cette notation ordonnée, la comparabilité est une relation d'équivalence : pour toutes suites $(u_n)_{n \in \N}$, $(v_n)_{n \in \N}$ et $(w_n)_{n \in \N}$~:
    \begin{itemize}
        \item $u_n \symbolelim{n \to + \infty}{}{=} \Theta(u_n)$
        \item $u_n \symbolelim{n \to + \infty}{}{=} \Theta(v_n) \Leftrightarrow v_n \symbolelim{n \to + \infty}{}{=} \Theta(u_n)$
        \item $\Big(u_n \symbolelim{n \to + \infty}{}{=} \Theta(v_n) \quad \text{et}\quad v_n \symbolelim{n \to + \infty}{}{=} \Theta(w_n) \Big) \implies u_n \symbolelim{n \to + \infty}{}{=} \Theta(w_n)$
    \end{itemize}
\end{remarque}

\begin{definition}{}{équivalence asymptotique}
    Soit $(u_n)_{n \in \N}$ et $(v_n)_{n \in \N}$ deux suites réelles positives. On dit que \notion{$(u_n)_{n \in \N}$ est équivalente à $v_n$} si~:
    $$\lim_{n\to + \infty} \frac{u_n}{v_n} = 1$$
    On note alors $\displaystyle u_n \symbolelim{n \to + \infty}{}{\sim} v_n$
\end{definition}

\begin{remarque}{}{cruciale}
    Si l'équivalence entre deux suites implique leur comparabilité (en prenant $C_1 = C_2 = 1$), la réciproque est fausse.

\end{remarque}

\begin{theoreme}{}{lemme de sommation}
    Soit $g : \R_+ \to \R_+$ une fonction continue et intégrable vérifiant~:
    \begin{enumeratebf}
        \item $g$ est croissante
        \item toute primitive $G$ de $g$ vérifie~:
        \begin{itemize}
            \item  $G(n) \symbolelim{n \to + \infty}{}{=} \Theta\Big(G(n+1)\Big)$
            \item $G(n) \flechelim{n \to + \infty}{} +\infty$
        \end{itemize}
    \end{enumeratebf}
    Alors les suites solutions de $u_n = g(n) + u_{n-1}$ sont en $\Theta\Big(G(n)\Big)$ avec $G$ une primitive de $g$.
\end{theoreme}

\begin{remarque}{}{concernant l'énoncé}
    Les conditions sont vérifiées dans le cas où $g$ est polynomiale, exponentielle ou bien encore logarithmique, et ce seront les seuls cas que nous étudierons dans un contexte d'étude de complexité.
    En particulier, supposer que $g = \mc{O}(K^n)$ comme le fait M. LAPOIRE donne l'hypothèse \textit{2}, tandis que la croissance de $g$ est assurée par sa nature de complexité.
    
\end{remarque}

\begin{demonstrationbreak}
    Soit $(u_n)_{n \in \N}$ solution. On remarque déjà que~:
    $$u_n = g(n) + g(n-1) + u_{n-2} = \sum_{i=1}^{n} g(i) + u_0$$
    Montrons alors qu'il existe $C_1 > 0$ et $C_2 > 0$ tels que~:
    $$C_1 G(n) \leq u_n \leq C_2 G(n)$$
    Soit $i \in \N^*$ et $x \in \seg{i}{i+1}$. Comme $g$ est croissante, on a~:
    $$g(x-1) \leq g(i) \leq g(x)$$
    En intégrant entre $i$ et $i+1$, on obtient par croissance de l'intégrale~:
    $$\int_{i}^{i+1} g(x-1) \d{x} \leq g(i) \leq \int_{i}^{i+1} g(x) \d{x}$$
    En sommant entre $i=1$ et $i=n$, on a par relation de Chasles~:
    $$\int_{0}^{n} g(x) \d{x} \leq \sum_{i=1}^{n} g(i) \leq \int_{1}^{n+1} g(x) \d{x}$$
    Soit~:
    $$ G(n) - G(0) \leq u_n - u_{0} \leq G(n+1) - G(1)$$
    Mais encore~:
    $$ \underbrace{\frac{G(n) - G(0) + u_{0}}{G(n)}}_{A_n}G(n) \leq u_{n} \leq \underbrace{\frac{G(n+1) - G(1) + u_{0}}{G(n)}}_{B_n}G(n)$$
    \begin{itemize}
        \item On a $A_n \flechelim{n \to + \infty}{} 1$ par hypothèse \textit{2.2}. La suite $(A_n)_{n \in \N}$ est convergente, donc bornée.
        \item On a $G(n+1) \symbolelim{n \to + \infty}{}{=} \Theta\Big(G(n)\Big)$. Ainsi, il existe $K_1$ et $K_2$ dans $\R_+^*$ telles que~:
        $$K_1 \leq \frac{G(n+1)}{G(n)} \leq K_2$$
        Puis $\displaystyle\frac{u_0 - G(1)}{G(n)} \flechelim{n\to + \infty}{}0$. Par théorème de conservation des inégalités larges par passage à la limite, on obtient~:
        $$K_1 \leq B_n \leq K_2$$
    \end{itemize}
    Ces fractions $A_n$ et $B_n$ sont toutes les deux bornées pour les raisons qui précèdent. On en déduit l'existence de $C_1$ et $C_2$.
\end{demonstrationbreak}
