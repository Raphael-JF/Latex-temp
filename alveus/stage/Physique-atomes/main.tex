\section{Structure de la matière}

\subsection{De quoi est faite la matière ?}

\begin{itemize}
    \item Toute la matière est constituée de particules extrêmement petites appelées \notion{atomes}.
    \item Un atome est électriquement neutre.
    \item Les atomes sont les "briques élémentaires" qui composent toute la matière (air, eau, métaux, êtres vivants…).
\end{itemize}

\subsection{Structure d’un atome}

\begin{itemize}
    \item Un atome est constitué :
    \begin{itemize}
        \item d’un \notion{noyau} central,
        \item d’\notion{électrons} qui gravitent autour du noyau.
    \end{itemize}
    
    \item Le noyau contient :
    \begin{itemize}
        \item des \notion{protons} (chargés positivement),
        \item des \notion{neutrons} (sans charge).
    \end{itemize}
    
    \item Les électrons sont chargés négativement.
\end{itemize}

\subsection{Les éléments chimiques}

\begin{itemize}
    \item Un \notion{élément chimique} est défini par le nombre de protons dans le noyau.
    \item Chaque élément possède un \notion{symbole chimique} :
    \begin{itemize}
        \item H : hydrogène
        \item O : oxygène
        \item C : carbone
        \item Fe : fer
    \end{itemize}
\end{itemize}

\section{Molécules}

\subsection{Définition}

\begin{itemize}
    \item Une \notion{molécule} est un assemblage de plusieurs atomes liés entre eux.
    \item Les atomes s’assemblent pour former des substances nouvelles.
\end{itemize}

\subsection{Exemples de molécules}

\begin{itemize}
    \item L’eau est constituée de deux atomes d’hydrogène et d’un atome d’oxygène : \(\mathrm{H_2O}\).
    \item Le dioxygène est constitué de deux atomes d’oxygène : \(\mathrm{O_2}\).
    \item Le dioxyde de carbone est constitué d’un atome de carbone et de deux atomes d’oxygène : \(\mathrm{CO_2}\).
\end{itemize}

\begin{exemple}{}{}
    L’air est un mélange de plusieurs gaz, principalement du dioxygène (\(\mathrm{O_2}\)) et du diazote (\(\mathrm{N_2}\)).
\end{exemple}

\section{Transformations chimiques}

\subsection{Réaction chimique}

\begin{itemize}
    \item Lors d’une \notion{réaction chimique}, les atomes se réorganisent.
    \item Les atomes sont conservés : ils ne disparaissent pas et ne sont pas créés.
    \item Les molécules changent, mais les atomes restent les mêmes.
\end{itemize}

\subsection{Conservation de la masse}

\begin{itemize}
    \item Au cours d’une transformation chimique, la masse totale se conserve.
\end{itemize}
