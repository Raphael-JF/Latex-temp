\begin{exercice}{}{}
    Voici des informations sur une fonction $h$:
    \begin{align*}
        \begin{array}{ll}
        h(-3) = 3 &\quad h(0) = -2 \\
        h(-1) = 2 &\quad h(1) = -2 \\
        h(3) = 2 &\quad h(5) = 0
        \end{array}
    \end{align*}
    Quelle est l'image par la fonction $h$ des nombres $-3$, $0$ et $5$~?:
    Citer un antécédent par la fonction $h$ des nombres $-2$ et $3$~:
    Citer un nombre dont l'image par $h$ est $2$.
\end{exercice}
