\section{Développement}

\begin{multicols}{2}
\begin{exercice}{}{}
    Développer et réduire les expressions algébriques suivantes~:
    \begin{enumeratebf}
        \item $ A = (4x+3)(7x-2)-2x$
        \item $B = (x+3)(2x-1) + x-1$
        \item $C = x + 5(2x-3) + 8(3-2x)$
        \item $D = 2(2+x) - 5(2x -3)$
    \end{enumeratebf}
\end{exercice}


\newcolumn

\begin{exercice}{}{des carrés}
    Développer et réduire les expressions algébriques suivantes~:
    \begin{enumeratebf}
        \item $A = (x-1)^2 - (2x+3)^2$
        \item $B = x^2 - (x-5)^2$
        \item $C = (3x-1)^2 + (3+x)^2$
        \item $D = (4x+1)(x-2)$
        \item $E = (x-2)(x+1)(x-3)$
    \end{enumeratebf}
\end{exercice}
\end{multicols}

\section{Factorisation}

\begin{multicols}{2}
\begin{exercice}{}{}
    Factoriser les expressions algébriques suivantes~:
    \begin{enumeratebf}
    \item $A = x^2 - 6x$
    \item $B = 2x^2 + 6x^3$
    \item $C = 25x^2 - 16$
    \item $D = (2x + 1)^2 - (x - 1)^2$
    \item $E = x^2 + 9 + 6x$
    \item $F = 4x^2 + 12x + 9$
    \item $G = 2x - 5 + (2x - 5)^2$
    \item $H = x^2 - 9 + (x - 3)(x + 1)$
\end{enumeratebf}
\end{exercice}


\begin{exercice}{}{des racines}
    Factoriser les expressions algébriques suivantes~:
    \begin{enumeratebf}
    \item $A = (2x + 1)^2 - (3x - 4)^2$
    \item $B = (3x - 2)^2 - (x + 1)^2$
    \item $C = (x + \sqrt{2})^2 - (3x + \sqrt{3})^2$
    \item $D = (-x + \sqrt{5})^2 - (2x - \sqrt{5})^2$
    \end{enumeratebf}
\end{exercice}
\newcolumn

\begin{exercice}{}{recherche des 0}
    Dans chacun des cas, résoudre l’équation $A = 0$.

\begin{enumeratebf}
    \item $A = (2x - 3)^2 - (x + 2)^2$
    \item $A = (x - 1)^2 - 9$
    \item $A = 4x^2 - 9$
    \item $A = (x + 1)^2 - (4x + 1)^2$
\end{enumeratebf}

\end{exercice}

\begin{exercice}{}{résolution d'équations}
    Résoudre les équations suivantes~:
    \begin{enumeratebf}
        \item $4x^2 - 12x + 9 = 9$
        \item $16x^2 + 16x + 4 = 4$
        \item $(x-3)^2 = 1$
        \item $(2x+1)^2 = 1$
        \item $(2x-1)(x+3) = -3$
        \item $(x+1)(5x-2) = -2$
    \end{enumeratebf}
\end{exercice}

\end{multicols}
