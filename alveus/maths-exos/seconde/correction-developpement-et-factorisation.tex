\section{Développement}

\begin{exercice}{}{}
    Développer et réduire les expressions algébriques suivantes~:
    \begin{enumeratebf}
        \item $ A = (4x+3)(7x-2)-2x = 28x^2 + 11x - 6$
        \item $B = (x+3)(2x-1) + x-1 = 2x^2 + 6x - 4$
        \item $C = x + 5(2x-3) + 8(3-2x) = -7x + 9 $
        \item $D = 2(2+x) - 5(2x -3) = -8x + 19$
    \end{enumeratebf}
\end{exercice}

\begin{exercice}{}{des carrés}
    Développer et réduire les expressions algébriques suivantes~:
    \begin{enumeratebf}
        \item $A = (x-1)^2 - (2x+3)^2 = -3x^2 - 14x-8$
        \item $B = x^2 - (x-5)^2 = 10x - 25$
        \item $C = (3x-1)^2 + (3+x)^2 = 10x^2 + 10$
        \item $D = (4x+1)(x-2) = 4x^2 - 7x -2$
        \item $E = (x-2)(x+1)(x-3) = x^3 - 4x^2 + x + 6$
    \end{enumeratebf}
\end{exercice}

\section{Factorisation}

\begin{exercice}{}{}
    \begin{enumerate}
    \item $A = x^2 - 6x = x(x - 6)$
    \item $B = 2x^2 + 6x^3 = 2x^2(1 + 3x)$
    \item $C = 25x^2 - 16 = (5x - 4)(5x + 4)$
    \item $D = (2x + 1)^2 - (x - 1)^2 = (x + 2)(3x)$
    \item $E = x^2 + 9 + 6x = (x + 3)^2$
    \item $F = 4x^2 + 12x + 9 = (2x + 3)^2$
    \item $G = 2x - 5 + (2x - 5)^2 = (2x - 5)(2x - 4)$
    \item $H = x^2 - 9 + (x - 3)(x + 1) = 2x(x - 1)$
\end{enumerate}
\end{exercice}

\begin{exercice}{}{des racines}
    Factoriser les expressions algébriques suivantes~:
    \begin{enumeratebf}
    \item $A = (2x + 1)^2 - (3x - 4)^2 = \big((2x + 1) - (3x - 4)\big)\big((2x + 1) + (3x - 4)\big) = (5 - x)(5x - 3)$
    \item $B = (3x - 2)^2 - (x + 1)^2 = \big((3x - 2) - (x + 1)\big)\big((3x - 2) + (x + 1)\big) = (2x - 3)(4x - 1)$
    \item $C = (x + \sqrt{2})^2 - (3x + \sqrt{3})^2 = \big((x + \sqrt{2}) - (3x + \sqrt{3})\big)\big((x + \sqrt{2}) + (3x + \sqrt{3})\big) = (-2x + \sqrt{2} - \sqrt{3})(4x + \sqrt{2} + \sqrt{3})$
    \item $D = (-x + \sqrt{5})^2 - (2x - \sqrt{5})^2 = \big((-x + \sqrt{5}) - (2x - \sqrt{5})\big)\big((-x + \sqrt{5}) + (2x - \sqrt{5})\big) = (-3x + 2\sqrt{5})(x)$
\end{enumeratebf}

\end{exercice}

\begin{exercice}{}{recherche des 0}
    Dans chacun des cas, résoudre l’équation $A = 0$.


\begin{enumerate}
    \item $A = (2x - 3)^2 - (x + 2)^2 = 0 \iff x \in \{5, \frac{1}{3}\}$
    \item $A = (x - 1)^2 - 9 = \iff x \in \{-2,4\}$
    \item $A = 4x^2 - 9 = \iff x \in \{-\frac{3}{2}, \frac{3}{2}\}$
    \item $A = (x + 1)^2 - (4x + 1)^2 = \iff x \in \{0, - \frac{2}{5}\}$
\end{enumerate}
\end{exercice}

\begin{exercice}{}{résolution d'équations}
    Résoudre les équations suivantes~:
    \begin{enumeratebf}
        \item $4x^2 - 12x + 9 = 9 \iff x\in\{0, 3\}$
        \item $16x^2 + 16x + 4 = 4 \iff x \in \{0, -1\}$
        \item $(x-3)^2 = 1 \iff x \in \{2, 4\}$
        \item $(2x+1)^2 = 1 \iff x \in \{0, -1\}$
        \item $(2x-1)(x+3) = -3 \iff x \in \{0, -\frac{5}{2}\}$
        \item $(x+1)(5x-2) = -2 \iff x \in \{0, -\frac{3}{5}\}$
    \end{enumeratebf}
\end{exercice}
